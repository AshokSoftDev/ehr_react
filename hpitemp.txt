import { toothImages } from "@/assets/toothImages";
import { useState } from "react";

type DentitionView = "primary" | "mixed" | "permanent";

const PERMANENT_UPPER = [
  "18",
  "17",
  "16",
  "15",
  "14",
  "13",
  "12",
  "11",
  "21",
  "22",
  "23",
  "24",
  "25",
  "26",
  "27",
  "28",
] as const;

const PERMANENT_LOWER = [
  "48",
  "47",
  "46",
  "45",
  "44",
  "43",
  "42",
  "41",
  "31",
  "32",
  "33",
  "34",
  "35",
  "36",
  "37",
  "38",
] as const;

const PRIMARY_UPPER = [
  "55",
  "54",
  "53",
  "52",
  "51",
  "61",
  "62",
  "63",
  "64",
  "65",
] as const;

const PRIMARY_LOWER = [
  "85",
  "84",
  "83",
  "82",
  "81",
  "71",
  "72",
  "73",
  "74",
  "75",
] as const;

const MIXED_LAYOUT = {
  permanentUpper: PERMANENT_UPPER,
  permanentLower: PERMANENT_LOWER,
  primaryUpper: PRIMARY_UPPER,
  primaryLower: PRIMARY_LOWER,
} as const;

interface ToothProps {
  toothNumber: string;
}

function Tooth({ toothNumber }: ToothProps) {
  const imageSrc = toothImages[toothNumber];

  return (
    <div className="flex w-10 flex-col items-center">
      {imageSrc ? (
        <img
          src={imageSrc}
          alt={`Tooth ${toothNumber}`}
          className="h-10 w-10 object-contain"
        />
      ) : (
        <div className="h-10 w-10 rounded bg-gray-100" />
      )}
      <span className="mt-1 text-xs font-medium text-gray-700">
        {toothNumber}
      </span>
    </div>
  );
}

function ArchLabel({ children }: { children: string }) {
  return (
    <div className="text-center text-[11px] font-semibold uppercase tracking-wide text-gray-500">
      {children}
    </div>
  );
}

const PERMANENT_SLOT_COUNT = PERMANENT_UPPER.length;

function renderArch(teeth: readonly string[]) {
  return (
    <div className="flex justify-center space-x-1">
      {teeth.map((tooth) => (
        <Tooth key={tooth} toothNumber={tooth} />
      ))}
    </div>
  );
}

function renderAlignedPrimaryArch(teeth: readonly string[]) {
  const totalSlots = PERMANENT_SLOT_COUNT;
  const missing = totalSlots - teeth.length;
  const leftSpacers = Math.floor(missing / 2);
  const rightSpacers = missing - leftSpacers;

  return (
    <div className="flex justify-center space-x-1">
      {Array.from({ length: leftSpacers }).map((_, index) => (
        <div key={`left-spacer-${index}`} className="h-10 w-10" />
      ))}

      {teeth.map((tooth) => (
        <Tooth key={tooth} toothNumber={tooth} />
      ))}

      {Array.from({ length: rightSpacers }).map((_, index) => (
        <div key={`right-spacer-${index}`} className="h-10 w-10" />
      ))}
    </div>
  );
}

interface DentitionToggleProps {
  view: DentitionView;
  onChange: (view: DentitionView) => void;
}

function DentitionToggle({ view, onChange }: DentitionToggleProps) {
  const options: { value: DentitionView; label: string }[] = [
    { value: "primary", label: "Primary Dentition" },
    { value: "mixed", label: "Mixed Dentition" },
    { value: "permanent", label: "Permanent Dentition" },
  ];

  return (
    <div className="flex flex-wrap items-center gap-4">
      {options.map((option) => {
        const selected = view === option.value;

        return (
          <label
            key={option.value}
            className="flex cursor-pointer items-center gap-2"
          >
            <span
              className={`flex h-4 w-4 items-center justify-center rounded-full border-2 ${
                selected ? "border-orange-500" : "border-gray-300"
              }`}
            >
              {selected && (
                <span className="h-2 w-2 rounded-full bg-orange-500" />
              )}
            </span>
            <input
              type="radio"
              className="sr-only"
              checked={selected}
              onChange={() => onChange(option.value)}
            />
            <span
              className={`text-sm ${
                selected ? "font-semibold text-orange-500" : "text-gray-700"
              }`}
            >
              {option.label}
            </span>
          </label>
        );
      })}
    </div>
  );
}

export function HPIDentalChart() {
  const [view, setView] = useState<DentitionView>("permanent");

  return (
    <div className="rounded-md border-y-4 border-orange-500 bg-white p-4 shadow-md">
      <div className="flex flex-col gap-3">
        <div className="flex items-center justify-between gap-2">
          <h3 className="text-sm font-semibold text-gray-800">Dental Chart</h3>
        </div>

        <DentitionToggle view={view} onChange={setView} />

        {view === "permanent" && (
          <div className="mt-4 space-y-3">
            {/* <ArchLabel>Upper Arch</ArchLabel> */}
            {renderArch(PERMANENT_UPPER)}

            {/* <ArchLabel>Lower Arch</ArchLabel> */}
            {renderArch(PERMANENT_LOWER)}
          </div>
        )}

        {view === "primary" && (
          <div className="mt-4 space-y-3">
            {/* <ArchLabel>Primary Upper Arch</ArchLabel> */}
            {renderAlignedPrimaryArch(PRIMARY_UPPER)}

            {/* <ArchLabel>Primary Lower Arch</ArchLabel> */}
            {renderAlignedPrimaryArch(PRIMARY_LOWER)}
          </div>
        )}

        {view === "mixed" && (
          <div className="mt-4 space-y-3">
            {/* <ArchLabel>Permanent Upper Arch</ArchLabel> */}
            {renderArch(MIXED_LAYOUT.permanentUpper)}

            {/* <ArchLabel>Primary Upper Arch</ArchLabel> */}
            {renderAlignedPrimaryArch(MIXED_LAYOUT.primaryUpper)}

            {/* <ArchLabel>Primary Lower Arch</ArchLabel> */}
            {renderAlignedPrimaryArch(MIXED_LAYOUT.primaryLower)}
          </div>
        )}
      </div>
    </div>
  );
}

export default HPIDentalChart;
